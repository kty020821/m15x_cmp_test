def _f(v): return round(float(v), 3) if v is not None else None def _box_stats_sql(cols, from_where): """지정 컬럼들의 min/q1/median/q3/max/avg 를 한 번에 계산하는 SQL 생성""" parts = [] for c in cols: parts += [ f'MIN("{c}")', f'PERCENTILE_CONT(0.25) WITHIN GROUP (ORDER BY "{c}")', f'PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY "{c}")', f'PERCENTILE_CONT(0.75) WITHIN GROUP (ORDER BY "{c}")', f'MAX("{c}")', f'AVG("{c}")', ] return f'SELECT {", ".join(parts)} FROM {from_where}' def _unpack_box(row, cols): """_box_stats_sql 결과 한 행을 컬럼별 dict 로 분해""" out, j = {}, 0 for c in cols: out[c] = { 'min': _f(row[j]), 'q1': _f(row[j + 1]), 'med': _f(row[j + 2]), 'q3': _f(row[j + 3]), 'max': _f(row[j + 4]), 'avg': _f(row[j + 5]), } j += 6 return out
